  var COVER_SECONDS = 2;   // ← 2026-09-12 Anita 改成 2 秒（原本 5 秒）

  /* 把 VIDEOS 裡的值變成真正的網址。
     · 填 'k50s-0912.mp4'（只有檔名）→ 自動接在 BASE 後面 ← **平常都用這種**
     · 填 'https://…'（完整網址）   → 原樣使用 ← 檔案不在我們自己的 repo 時的逃生口
     兩種混用也可以。 */
  function url(u){
    u = (u || '').trim();
    if (!u) return '';
    return /^(https?:)?\/\//i.test(u) ? u : BASE + u.replace(/^\/+/, '');
  }

  var row      = root.querySelector('.lnihm-vid-row');
  var dotsWrap = root.querySelector('.lnihm-vid-dots');
  var foot     = root.querySelector('.lnihm-vid-foot');
  var modal    = document.getElementById('lnihmVidModal');
  var video    = modal.querySelector('.lnihm-vid-v');
  var btnX     = modal.querySelector('.lnihm-vid-x');
  var btnSound = modal.querySelector('.lnihm-vid-sound');
  var seek     = modal.querySelector('.lnihm-vid-seek');
  var elNow    = modal.querySelector('.lnihm-vid-now');
  var elDur    = modal.querySelector('.lnihm-vid-dur');
  var buy      = modal.querySelector('.lnihm-vid-buy');
  var buyText  = buy.querySelector('span');
  var elH      = modal.querySelector('.lnihm-vid-h');
  var elP      = modal.querySelector('.lnihm-vid-p');

  var PLAY = '<svg class="lnihm-vid-play" viewBox="0 0 24 24" aria-hidden="true"><circle cx="12" cy="12" r="11" fill="rgba(255,255,255,.92)"/><path d="M9.8 8.2v7.6l6.2-3.8z" fill="#3c3c3c"/></svg>';
  var ICON_MUTED = '<path d="M4 9v6h4l5 5V4L8 9H4zm13.5 3a4.5 4.5 0 0 0-1.6-3.4v6.8A4.5 4.5 0 0 0 17.5 12zM3 3.3 20.7 21 19.3 22.4 1.6 4.7 3 3.3z"/>';
  var ICON_ON    = '<path d="M4 9v6h4l5 5V4L8 9H4zm11.5 3a4.5 4.5 0 0 0-2.5-4v8a4.5 4.5 0 0 0 2.5-4zM13 1.8v2.1a8.1 8.1 0 0 1 0 16.2v2.1A10.2 10.2 0 0 0 13 1.8z"/>';

  var reduceMotion = !!(window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches);

  /* 首頁封面：只播「現在看得見」的那幾支。threshold 0.25 ＝ 露出四分之一就開始播，
     手機一屏 2.3 張時，正在滑進來的那張會先動起來，不會等到完全停好才播。 */
  var io = ('IntersectionObserver' in window) ? new IntersectionObserver(function(entries){
    for (var i = 0; i < entries.length; i++){
      var el = entries[i].target;
      if (entries[i].isIntersecting){
        /* 捲回來時從頭再播一次 2 秒（不是停在上次停住的那一格）。
           ⚠️ readyState >= 1 才動 currentTime：preload:'none' 的影片還沒讀到長度，
              這時候設 currentTime 會被忽略或報錯。 */
        try { if (el.readyState >= 1 && el.currentTime >= COVER_SECONDS) el.currentTime = 0; } catch(e){}
        var pr = el.play();
        if (pr && pr.catch) pr.catch(function(){});   // 被瀏覽器擋下就算了，封面不是必要功能
      } else {
        try { el.pause(); } catch(e){}
      }
    }
  }, { threshold: 0.25 }) : null;
  function isVideoFile(u){ return /\.(mp4|webm|mov)(\?|$)/i.test(u || ''); }
  function mmss(s){
    if (!isFinite(s) || s < 0) s = 0;
    var m = Math.floor(s / 60), x = Math.floor(s % 60);
    return m + ':' + (x < 10 ? '0' : '') + x;
  }
  function addImg(th, src){
    var im = document.createElement('img');
    im.src = src;
    im.alt = '';                         // 裝飾性：標題已在下方，讀屏不必唸兩次
    im.setAttribute('loading', 'lazy');
    im.setAttribute('decoding', 'async');
    th.appendChild(im);
  }

  /* ---------------- 首頁卡片 ---------------- */
  function makeCard(v, idx){
    var b = document.createElement('button');
    b.type = 'button';
    b.className = 'lnihm-vid-card';
    b.setAttribute('data-i', String(idx));
    b.setAttribute('aria-haspopup', 'dialog');
    b.setAttribute('aria-label', (v.title || v.headline || '影片') + '，開啟影片');

    var th = document.createElement('div');
    th.className = 'lnihm-vid-thumb';

    if (v.gif && isVideoFile(v.gif)){
      if (reduceMotion){
        /* 使用者在系統設定開了「減少動態效果」→ 短片不播。
           🔴 這時**不可以**退回 img src 指向 mp4——瀏覽器不會把 mp4 當圖片，
              結果是破圖，比不放還糟。改用靜態封面；沒有靜態封面就留灰底。
           ⚠️ GIF 沒有這個選項：瀏覽器沒有提供「暫停 GIF」的方法。這是選 GIF 的已知代價。 */
        if (v.poster) addImg(th, url(v.poster));
        else th.className += ' lnihm-vid-noimg';
      } else {
        var vd = document.createElement('video');
        vd.src = url(v.gif);
        /* 🔴 loop 關掉：封面只播 COVER_SECONDS 秒就停，不循環（循環的話 loop 會先跳回 0）。 */
        vd.muted = true; vd.loop = false; vd.playsInline = true;
        vd.addEventListener('timeupdate', function(){
          if (vd.currentTime >= COVER_SECONDS) { try { vd.pause(); } catch(e){} }
        });
        vd.setAttribute('muted', '');        // 🔴 屬性與 property 都要設：部分 iOS 版本只認屬性
        vd.setAttribute('playsinline', '');  // 🔴 少了它 iPhone 會強制全螢幕播放
        vd.setAttribute('aria-hidden', 'true');
        vd.setAttribute('tabindex', '-1');
        if (v.poster) vd.setAttribute('poster', url(v.poster));
        /* 🔴🔴 preload:'none' ＋ 只播「捲到看得見」的那幾張，不要用 autoplay。
           為什麼：封面可以直接填 mp4（跟浮層同一支檔，換片的人只要傳一個檔）。
           但那代表首頁最多會掛 20 支影片——**全部自動播＝一次下載 20 支**。
           一支壓過的直式短片約 8 MB，20 支就是 160 MB，而且手機吃的是行動網路。
           改成「看得見才播、離開畫面就暫停」之後，一開始只會載到畫面內的那 2–6 支。
           ⚠️ 舊版（爬部落格那版）本來就有這段，改寫浮層時被我拿掉了，這裡補回來。
           ⚠️ 沒有 IntersectionObserver 的舊瀏覽器退回 autoplay——**會全部載**，
              但那批瀏覽器占比極低，而且退化方向是「能看」不是「壞掉」。 */
        if (io) { vd.setAttribute('preload', 'none'); io.observe(vd); }
        else    { vd.autoplay = true; vd.setAttribute('preload', 'metadata'); }
        th.appendChild(vd);
      }
    } else if (v.gif){
      addImg(th, url(v.gif));
    } else {
      th.className += ' lnihm-vid-noimg';
    }

    var mask = document.createElement('div');
    mask.className = 'lnihm-vid-mask';
    th.appendChild(mask);
    th.insertAdjacentHTML('beforeend', PLAY);   // 固定字串·不含外部資料

    var t = document.createElement('div');
    t.className = 'lnihm-vid-title';
    /* 首頁封面下方顯示的字＝**敘述標題**（2026-09-12 Anita 指定）。
       title 是選填的短版覆寫，平常留空即可——留空就直接用 headline，
       首頁和浮層是同一句話，換片的人少填一個欄位。 */
    t.textContent = v.title || v.headline || '';   // 🔴 textContent 不是 innerHTML → 防注入
    b.appendChild(th); b.appendChild(t);
    return b;
  }

  function renderCards(){
    /* 🔴 沒有 mp4 的整筆跳過：浮層沒東西可播，放上去就是一張點了沒反應的卡 */
    var ok = VIDEOS.filter(function(v){ return v && v.mp4; });
    /* 🔴 超過上限是「靜靜切掉」，畫面上看不出來 → 至少在主控台留一句，
       否則換片的人會以為第 21 部沒存到 */
    if (ok.length > MAX){
      try { console.warn('[好物放映] 清單有 ' + ok.length + ' 部，超過上限 ' + MAX +
        ' 部，第 ' + (MAX + 1) + ' 部之後不會顯示。'); } catch(e){}
    }
    var list = ok.slice(0, MAX);
    row.innerHTML = '';
    list.forEach(function(v, i){ row.appendChild(makeCard(v, i)); });
    row._list = list;
    /* 目的地未定案前的暫時規則：影片少於 4 支就整顆隱藏（見上方按鈕的註解） */
    foot.hidden = list.length < 4;
    page = 0; build();
  }

  /* ---------------- 浮層 ---------------- */
  var lastFocus = null, dragging = false, savedScroll = 0;

  function setSound(on){
    video.muted = !on;
    btnSound.setAttribute('aria-pressed', on ? 'true' : 'false');
    btnSound.querySelector('svg').innerHTML = on ? ICON_ON : ICON_MUTED;
  }

  /* 浮層裡現在真的按得到的東西（順序＝Tab 走的順序）。
     🔴 一定要每次重算，不能寫死：商品連結在沒填 url 時會被 display:none，
        寫死的話 Tab 會停在一個看不見的東西上。 */
  function focusables(){
    var all = modal.querySelectorAll('button, input, a[href]');
    var out = [];
    for (var i = 0; i < all.length; i++){
      var el = all[i];
      if (el.offsetWidth || el.offsetHeight || el.getClientRects().length) out.push(el);
    }
    return out;
  }

  function openAt(i){
    var list = row._list || [];
    var v = list[i];
    if (!v || !v.mp4) return;

    video.src = url(v.mp4);
    video.poster = url(v.poster) || url(v.gif) || '';
    video.loop = true;
    video.currentTime = 0;
    seek.value = 0; elNow.textContent = '0:00'; elDur.textContent = '0:00';

    var p = v.product || {};
    if (p.url){
      buy.href = p.url;
      buyText.textContent = p.name || '查看商品';
      buy.style.display = '';
    } else {
      buy.style.display = 'none';   // 沒填就整顆收起來，不要留點不動的假按鈕
    }
    elH.textContent = v.headline || '';
    elP.textContent = v.desc || '';

    lastFocus = document.activeElement;
    modal.classList.add('lnihm-vid-open');
    /* 鎖背景：先記住現在捲到哪，再把 body 釘住（見 CSS 那段註解） */
    savedScroll = window.pageYOffset || document.documentElement.scrollTop || 0;
    document.body.style.top = (-savedScroll) + 'px';
    document.body.classList.add('lnihm-vid-lock');

    /* 🔴 先試「有聲音」播——浮層是使用者點擊打開的，屬於瀏覽器認可的使用者操作，
       所以帶聲音播放通常會被允許。
       ⚠️ 但 Safari 仍可能擋下，所以一定要接住失敗：退回靜音再播一次，
          並把喇叭按鈕誠實顯示成「靜音中」。
       🔑 這是改用自架 video 換到最大的一件事：play() 會回傳成功或失敗，
          **程式知道結果**。YouTube 的 postMessage 沒有回傳值，按鈕只能用猜的——
          前一版就是無條件把圖示切成「有聲」，於是圖示會說謊。 */
    setSound(true);
    var pr = video.play();
    if (pr && pr.catch){
      pr.catch(function(){
        setSound(false);
        var again = video.play();
        if (again && again.catch) again.catch(function(){ /* 連靜音都播不了：留 poster，使用者可按進度條 */ });
      });
    }
    btnX.focus();
  }

  function close(){
    modal.classList.remove('lnihm-vid-open');
    document.body.classList.remove('lnihm-vid-lock');
    document.body.style.top = '';
    window.scrollTo(0, savedScroll);   // 把 position:fixed 造成的跳頂捲回去
    /* 🔴 一定要暫停並把 src 清掉：只隱藏的話影片會繼續播、繼續發出聲音，
       而且繼續下載。removeAttribute('src') + load() 是釋放的標準做法。 */
    try { video.pause(); } catch(e){}
    video.removeAttribute('src');
    try { video.load(); } catch(e){}
    if (lastFocus && lastFocus.focus) lastFocus.focus();   // 焦點還給剛才那張卡
  }

  /* ---------------- 進度條 ---------------- */
  video.addEventListener('loadedmetadata', function(){
    elDur.textContent = mmss(video.duration);
  });
  video.addEventListener('timeupdate', function(){
    if (dragging || !isFinite(video.duration) || !video.duration) return;
    seek.value = String(Math.round(video.currentTime / video.duration * 1000));
    elNow.textContent = mmss(video.currentTime);
  });
  /* 拖曳中先不要讓 timeupdate 把把手拉回去，放開才真的跳 */
  seek.addEventListener('input', function(){
    dragging = true;
    if (isFinite(video.duration) && video.duration){
      elNow.textContent = mmss(seek.value / 1000 * video.duration);
    }
  });
  seek.addEventListener('change', function(){
    if (isFinite(video.duration) && video.duration){
      video.currentTime = seek.value / 1000 * video.duration;
    }

    if (!card || !row.contains(card)) return;

  modal.addEventListener('click', function(e){ if (e.target === modal) close(); });

  /* 🔴 把 Tab 關在浮層裡。少了這段，從最後一顆按鈕再按一次 Tab 就會跑到
     浮層「後面」的頁面元素（LINE 訂閱、Footer 一整排連結）——那些被 90% 黑的
     遮罩蓋住，所以**焦點框看不見，人不知道自己按到哪了**。
     ⚠️ aria-modal="true" 幫不上忙：它只管讀屏的瀏覽游標，管不到 Tab 鍵順序。
     ⚠️ 另一個常見寫法是對 body 的其他子元素加 inert，**在 SHOPLINE 上不能用**：
        浮層不是 body 的直接子元素（外面還有 .shopline-section），
        對那層加 inert 會連浮層自己一起鎖死，叉叉都按不動。 */
  modal.addEventListener('keydown', function(e){
    if (e.key !== 'Tab') return;
    var f = focusables();
    if (!f.length) return;
    var first = f[0], last = f[f.length - 1];
    if (e.shiftKey && document.activeElement === first){ e.preventDefault(); last.focus(); }
    else if (!e.shiftKey && document.activeElement === last){ e.preventDefault(); first.focus(); }
  });
  document.addEventListener('keydown', function(e){
    if (e.key === 'Escape' && modal.classList.contains('lnihm-vid-open')) close();
  });

  /* ---------------- 分頁點點（比照其他區塊） ---------------- */
  var page = 0, pv = 1, pages = 1, lock = false;
  function cards(){ return [].slice.call(row.querySelectorAll('.lnihm-vid-card')); }
  function step(cs){ return cs.length > 1 ? (cs[1].offsetLeft - cs[0].offsetLeft) : (cs[0] ? cs[0].offsetWidth : 1); }
  var twIv;
  function tween(el, to, dur, done){
    clearInterval(twIv); el.style.scrollSnapType = 'none';
    var start = el.scrollLeft, change = to - start, t0 = Date.now();
    if (Math.abs(change) < 1){ el.scrollLeft = to; el.style.scrollSnapType = ''; if (done) done(); return; }
    twIv = setInterval(function(){
      var q = Math.min(1, (Date.now() - t0) / dur);
      el.scrollLeft = start + change * (1 - Math.pow(1 - q, 3));
      if (q >= 1){ clearInterval(twIv); el.style.scrollSnapType = ''; if (done) done(); }
    }, 16);
  }
  function build(){
    var cs = cards(); if (!cs.length){ dotsWrap.innerHTML = ''; return; }
    /* 🔴 用「軌道有沒有可捲動的寬度」判斷要不要點點，不要用視窗寬度判斷——
       斷點怎麼改都不用同步改這裡，卡片少到一屏放得下時也會自動收起來。 */
    if (row.scrollWidth <= row.clientWidth + 2){
      pages = 1; page = 0; dotsWrap.innerHTML = ''; return;
    }
    /* 🔴 一頁塞得下幾張＝**完整看得到**幾張。算式 floor((軌道寬 + 一個 gap) / (卡寬 + gap))，
       那個「+ 一個 gap」不能省：n 張卡在軌道上只有 (n−1) 個 gap，
       直接用 軌道寬/(卡寬+gap) 會少算最後那個不存在的 gap，剛好填滿時會整整少算一張。
       ⚠️ 仍然是 floor 不是 round：手機 2.3 張要算成 2 張。 */
    var s = step(cs);
    var gp = cs.length > 1 ? (s - cs[0].offsetWidth) : 0;
    pv = Math.max(1, Math.floor((row.clientWidth + gp) / s));
    pages = Math.ceil(cs.length / pv);
    dotsWrap.innerHTML = '';
    if (pages > 1){
      for (var i = 0; i < pages; i++){
        var b = document.createElement('button');
        b.type = 'button'; b.setAttribute('aria-label', '第 ' + (i + 1) + ' 頁');
        (function(idx){ b.addEventListener('click', function(){ go(idx); }); })(i);
        dotsWrap.appendChild(b);
      }
    }
    page = Math.min(page, pages - 1); mark();
  }
  function pageLeft(p){
    var cs = cards(); if (!cs.length) return 0;
    var idx = Math.min(p * pv, cs.length - 1);
    var maxSc = row.scrollWidth - row.clientWidth;
    return Math.max(0, Math.min(cs[idx].offsetLeft - cs[0].offsetLeft, maxSc));
  }
  function nearestPage(){
    var best = 0, bd = Infinity;
    for (var i = 0; i < pages; i++){
      var d = Math.abs(row.scrollLeft - pageLeft(i));
      if (d < bd){ bd = d; best = i; }
    }
    return best;
  }
  function mark(){
    var bs = dotsWrap.querySelectorAll('button');
    for (var i = 0; i < bs.length; i++){
      if (i === page) bs[i].setAttribute('aria-current', 'true');
      else bs[i].removeAttribute('aria-current');
    }
  }
  function go(p){
    page = Math.max(0, Math.min(p, pages - 1));
    lock = true; mark();
    tween(row, pageLeft(page), 380, function(){ lock = false; });
  }
  var sIv;
  row.addEventListener('scroll', function(){
    if (lock) return;
    clearTimeout(sIv);
    /* 用 setTimeout 節流，不要用 requestAnimationFrame：
       背景分頁的 rAF 完全不執行，旗標會永久卡住而且不報錯（既有踩雷紀錄）。 */
    sIv = setTimeout(function(){ page = nearestPage(); mark(); }, 90);
  });
  var rIv;
  window.addEventListener('resize', function(){ clearTimeout(rIv); rIv = setTimeout(build, 150); });

  renderCards();
})();
</script>
