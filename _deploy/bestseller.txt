<script>(function(){
  var root = document.getElementById('lnihmHot');
  if (!root || root.getAttribute('data-lnihm-init')) return;
  root.setAttribute('data-lnihm-init', '1');
  var row = root.querySelector('.lnihm-hot-row');
  var dotsWrap = root.querySelector('.lnihm-hot-dots');

  var SRC = '/categories/bestseller';
  var KEY = 'lnihm_hot_cache_v1';

  var TTL = 720 * 60 * 1000;

  var MAX = 12;

  function money(n){ return 'NT$' + String(n).replace(/\B(?=(\d{3})+(?!\d))/g, ','); }

  function makeCard(p){
    var a = document.createElement('a');
    a.className = 'lnihm-hot-card';
    a.setAttribute('href', p.url);
    var photo = document.createElement('div');
    photo.className = 'lnihm-hot-photo';
    if (p.img) photo.style.backgroundImage = "url('" + p.img.replace(/'/g, '%27') + "')";
    if (p.label){
      var tag = document.createElement('span');
      tag.className = 'lnihm-hot-tag';
      tag.textContent = p.label;
      photo.appendChild(tag);
    }
    if (p.save > 0){
      var save = document.createElement('span');
      save.className = 'lnihm-hot-save';
      save.textContent = '省 ' + money(p.save);
      photo.appendChild(save);
    }
    var body = document.createElement('div');
    body.className = 'lnihm-hot-body';
    var title = document.createElement('div');
    title.className = 'lnihm-hot-title';
    title.textContent = p.title || '';
    var prices = document.createElement('div');
    prices.className = 'lnihm-hot-prices';
    var now = document.createElement('span');
    now.className = 'lnihm-hot-now';
    now.textContent = money(p.sale);

    if (p.rng){
      var em = document.createElement('em');
      em.textContent = '起';
      now.appendChild(em);
    }
    prices.appendChild(now);
    if (p.reg && p.reg > p.sale){
      var was = document.createElement('span');
      was.className = 'lnihm-hot-was';
      was.textContent = money(p.reg);
      prices.appendChild(was);
    }
    body.appendChild(title); body.appendChild(prices);
    a.appendChild(photo); a.appendChild(body);
    return a;
  }

  var page = 0, pv = 1, pages = 1, lock = false;
  function cards(){ return [].slice.call(row.querySelectorAll('.lnihm-hot-card')); }
  function step(cs){ return cs.length > 1 ? (cs[1].offsetLeft - cs[0].offsetLeft) : (cs[0] ? cs[0].offsetWidth : 1); }
  var twIv;
  function tween(el, to, dur, done){
    clearInterval(twIv); el.style.scrollSnapType = 'none';
    var start = el.scrollLeft, change = to - start, t0 = Date.now();
    if (Math.abs(change) < 1){ el.scrollLeft = to; el.style.scrollSnapType = ''; if (done) done(); return; }
    twIv = setInterval(function(){
      var p = Math.min(1, (Date.now() - t0) / dur);
      el.scrollLeft = start + change * (1 - Math.pow(1 - p, 3));
      if (p >= 1){ clearInterval(twIv); el.style.scrollSnapType = ''; if (done) done(); }
    }, 16);
  }

  function build(){
    var cs = cards(); if (!cs.length) return;
    var s = step(cs); pv = Math.max(1, Math.floor(row.clientWidth / s));
    pages = Math.ceil(cs.length / pv);
    dotsWrap.innerHTML = '';
    if (pages > 1){
      for (var i = 0; i < pages; i++){
        var b = document.createElement('button');
        b.type = 'button'; b.setAttribute('aria-label', '第 ' + (i + 1) + ' 頁');
        (function(idx){ b.addEventListener('click', function(){ go(idx); }); })(i);
        dotsWrap.appendChild(b);
      }
    }
    page = Math.min(page, pages - 1); mark();
  }
  function pageLeft(p){
    var cs = cards(); if (!cs.length) return 0;
    var idx = Math.min(p * pv, cs.length - 1);
    var maxSc = row.scrollWidth - row.clientWidth;
    return Math.max(0, Math.min(cs[idx].offsetLeft - cs[0].offsetLeft, maxSc));
  }
  function nearestPage(){
    var best = 0, bd = Infinity;
    for (var i = 0; i < pages; i++){
      var d = Math.abs(row.scrollLeft - pageLeft(i));
      if (d < bd){ bd = d; best = i; }
    }
    return best;
  }
  function mark(){
    var bs = dotsWrap.querySelectorAll('button');
    for (var i = 0; i < bs.length; i++){
      if (i === page) bs[i].setAttribute('aria-current', 'true');
      else bs[i].removeAttribute('aria-current');
    }
  }
  function go(p){
    page = Math.max(0, Math.min(p, pages - 1));
    lock = true; mark();
    tween(row, pageLeft(page), 380, function(){ lock = false; });
  }
  var sIv;
  row.addEventListener('scroll', function(){
    if (lock) return;
    clearTimeout(sIv);
    sIv = setTimeout(function(){ page = nearestPage(); mark(); }, 90);
  });

  var down = false, sx = 0, sl = 0, moved = false;
  row.addEventListener('mousedown', function(e){
    down = true; moved = false; sx = e.pageX; sl = row.scrollLeft;
    row.classList.add('lnihm-grabbing');
  });
  window.addEventListener('mousemove', function(e){
    if (!down) return;
    var d = e.pageX - sx;
    if (Math.abs(d) > 3) moved = true;
    row.scrollLeft = sl - d;
    if (moved) e.preventDefault();
  });
  window.addEventListener('mouseup', function(e){
    if (!down) return;
    down = false; row.classList.remove('lnihm-grabbing');
    if (moved){ go(nearestPage()); }
  });
  row.addEventListener('click', function(e){ if (moved){ e.preventDefault(); moved = false; } }, true);
  var rIv;
  window.addEventListener('resize', function(){ clearTimeout(rIv); rIv = setTimeout(build, 150); });

  function render(list){
    if (!list || !list.length) return;
    row.innerHTML = '';
    list.slice(0, MAX).forEach(function(p){ row.appendChild(makeCard(p)); });
    page = 0; build();
  }

  function parse(html){
    var doc = new DOMParser().parseFromString(html, 'text/html');
    var items = doc.querySelectorAll('.productList__product');
    var seen = {}, out = [];
    for (var i = 0; i < items.length; i++){
      var el = items[i];
      var a = el.querySelector('a[href*="/products/"]');
      var t = el.querySelector('.title');
      if (!a || !t) continue;
      var url = (a.getAttribute('href') || '').split('?')[0];
      if (!url || seen[url]) continue;

      var priceEl = el.querySelector('.price__sale') || el.querySelector('.price__regular');
      if (!priceEl) continue;
      var priceTxt = (priceEl.textContent || '').replace(/,/g, '');
      var nums = priceTxt.match(/\d+/g);
      if (!nums || !nums.length) continue;
      var sale = parseInt(nums[0], 10);
      if (!sale) continue;

      var rng = (priceTxt.indexOf('~') >= 0) || nums.length > 1;

      var crossed = el.querySelector('.price-crossed');
      var reg = 0;
      if (crossed){
        var rn = (crossed.textContent || '').replace(/,/g, '').match(/\d+/);
        if (rn) reg = parseInt(rn[0], 10);
      }

      var save = (!rng && reg && reg > sale) ? (reg - sale) : 0;

      var labEl = el.querySelector('.product-customized-labels-content');
      var label = labEl ? (labEl.textContent || '').trim() : '';

      var img = '', imgEl = el.querySelector('img[data-srcset], img[srcset], img[data-src], img[src]');
      if (imgEl){
        var ss = imgEl.getAttribute('data-srcset') || imgEl.getAttribute('srcset') || '';
        var best = null, bestD = 1e9, re = /(https:\/\/[^\s,"]+?\/(\d+)x\.[a-z]+)\?/g, m2;
        while ((m2 = re.exec(ss)) !== null){
          var d = Math.abs(parseInt(m2[2], 10) - 720);
          if (d < bestD){ bestD = d; best = m2[1]; }
        }
        img = best || imgEl.getAttribute('data-src') || imgEl.getAttribute('src') || '';
      }
      if (!/^https:\/\//i.test(img)) continue;
      seen[url] = 1;
      out.push({ url: url, title: (t.textContent || '').trim(), sale: sale, rng: rng,
                 reg: reg, save: save, label: label, img: img });
    }

    return out;
  }

  function refresh(){
    fetch(SRC, { credentials: 'same-origin' })
      .then(function(r){ return r.ok ? r.text() : Promise.reject(r.status); })
      .then(function(html){
        var list = parse(html);
        if (list.length){
          render(list);
          try { localStorage.setItem(KEY, JSON.stringify({ t: Date.now(), list: list.slice(0, MAX) })); } catch(e){}
        }
      })
      .catch(function(){   });
  }

  build();

  var usedCache = false;
  try {
    var c = JSON.parse(localStorage.getItem(KEY) || 'null');
    if (c && c.list && c.list.length){
      render(c.list);
      usedCache = true;
      if (Date.now() - c.t > TTL) refresh();
    }
  } catch(e){}
  if (!usedCache) refresh();
})();
</script>
