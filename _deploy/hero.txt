    <script>const hero = document.getElementById('lnihmHero');
        const slides = hero.querySelectorAll('.lnihm-hero-slide');
        const dots = hero.querySelectorAll('.lnihm-hero-dot');
        let currentIndex = 0;
        function showSlide(index) {
            slides.forEach((slide, i) => {
                slide.classList.toggle('lnihm-hero-on', i === index);
                dots[i].classList.toggle('lnihm-hero-dot-on', i === index);
            });
            currentIndex = index;
        }
        function nextSlide() {
            showSlide((currentIndex + 1) % slides.length);
        }
        dots.forEach((dot, i) => {
            dot.addEventListener('click', () => showSlide(i));
        });
        setInterval(nextSlide, 5000);
</script>
