<script>(function(){
  var root=document.getElementById('lnihmPromo');
  if(!root||root.getAttribute('data-lnihm-init'))return;
  root.setAttribute('data-lnihm-init','1');
  var row=root.querySelector('.lnihm-promo-row');
  var dotsWrap=root.querySelector('.lnihm-promo-dots');
  if(!row||!dotsWrap)return;
  var page=0,pv=1,pages=1,lock=false,tmr;
  function cards(){ return [].slice.call(row.querySelectorAll('.lnihm-promo-card')); }
  function step(cs){ return cs.length>1 ? (cs[1].offsetLeft-cs[0].offsetLeft) : (cs[0]?cs[0].offsetWidth:1); }

  var twIv;
  function tween(el,to,dur,done){
    clearInterval(twIv); el.style.scrollSnapType='none';
    var start=el.scrollLeft, change=to-start, t0=Date.now();
    if(Math.abs(change)<1){ el.scrollLeft=to; el.style.scrollSnapType=''; if(done)done(); return; }
    twIv=setInterval(function(){
      var p=Math.min(1,(Date.now()-t0)/dur);
      el.scrollLeft=start+change*(1-Math.pow(1-p,3));
      if(p>=1){ clearInterval(twIv); el.style.scrollSnapType=''; if(done)done(); }
    },16);
  }
  function build(){
    var cs=cards(); if(!cs.length)return;
    var s=step(cs); pv=Math.max(1,Math.round(row.clientWidth/s));
    pages=Math.ceil(cs.length/pv);

    if(row.scrollWidth <= row.clientWidth + 2) pages = 1;
    dotsWrap.innerHTML='';
    if(pages>1){
      for(var i=0;i<pages;i++){
        var b=document.createElement('button');
        b.type='button'; b.setAttribute('aria-label','第 '+(i+1)+' 頁');
        (function(idx){ b.addEventListener('click',function(){ go(idx); }); })(i);
        dotsWrap.appendChild(b);
      }
    }
    page=Math.min(page,pages-1); mark();
  }

  var SPEED=250;

  function pageLeft(p){
    var cs=cards(); if(!cs.length)return 0;
    var idx=Math.min(p*pv, cs.length-1);
    var maxSc=row.scrollWidth-row.clientWidth;
    return Math.max(0,Math.min(cs[idx].offsetLeft-cs[0].offsetLeft, maxSc));
  }
  function nearestPage(){
    var best=0,bd=Infinity;
    for(var i=0;i<pages;i++){ var d=Math.abs(row.scrollLeft-pageLeft(i)); if(d<bd){bd=d;best=i;} }
    return best;
  }
  function go(p){
    p=Math.max(0,Math.min(p,pages-1)); page=p;
    lock=true; tween(row,pageLeft(p),SPEED,function(){ lock=false; });
    mark();
  }
  function mark(){ var k=dotsWrap.children; for(var i=0;i<k.length;i++) k[i].setAttribute('aria-current', i===page?'true':'false'); }
  function onScroll(){
    if(lock||pages<=1)return;
    var p=nearestPage();
    if(p!==page){ page=p; mark(); }
  }

  var THIRD=1/3;
  function snapByThird(){
    var cs=cards(); if(!cs.length)return;
    var s=step(cs), maxSc=row.scrollWidth-row.clientWidth, base=cs[0].offsetLeft;
    var delta=row.scrollLeft-dScroll;
    var a=Math.abs(delta), full=Math.floor(a/s), rem=a-full*s;
    var adv=full+(rem>=s*THIRD?1:0);
    var target=dScroll+(delta<0?-1:1)*adv*s;
    target=Math.max(0,Math.min(target,maxSc));
    var best=target,bd=Infinity;
    for(var i=0;i<cs.length;i++){
      var v=Math.max(0,Math.min(cs[i].offsetLeft-base,maxSc));
      var d=Math.abs(target-v);
      if(d<bd){bd=d;best=v;}
    }
    lock=true; tween(row,best,SPEED,function(){ lock=false; page=nearestPage(); mark(); });
  }
  var st;
  row.addEventListener('scroll',function(){ clearTimeout(st); st=setTimeout(onScroll,60); },{passive:true});
  var rt;
  window.addEventListener('resize',function(){ clearTimeout(rt); rt=setTimeout(build,150); });

  var dDown=false,dMoved=false,dX=0,dScroll=0;
  var DRAG_MIN=8;
  function onMove(e){
    if(!dDown||e.pointerType!=='mouse')return;
    var dx=e.clientX-dX;
    if(Math.abs(dx)>DRAG_MIN)dMoved=true;
    if(dMoved){ row.scrollLeft=dScroll-dx; e.preventDefault(); }
  }
  function dEnd(){
    if(!dDown)return;dDown=false;
    window.removeEventListener('pointermove',onMove);
    window.removeEventListener('pointerup',dEnd);
    window.removeEventListener('pointercancel',dEnd);
    row.classList.remove('lnihm-grabbing');

    if(dMoved) snapByThird();
    else row.style.scrollSnapType='';
  }
  row.addEventListener('pointerdown',function(e){
    if(e.pointerType!=='mouse'||e.button!==0)return;
    dDown=true;dMoved=false;dX=e.clientX;dScroll=row.scrollLeft;
    row.style.scrollSnapType='none';row.classList.add('lnihm-grabbing');
    window.addEventListener('pointermove',onMove);
    window.addEventListener('pointerup',dEnd);
    window.addEventListener('pointercancel',dEnd);
  });

  row.addEventListener('click',function(e){ if(dMoved){e.preventDefault();e.stopPropagation();dMoved=false;} },true);
  build();
})();
</script>
