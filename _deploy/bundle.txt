<script>(function(){
  var root = document.getElementById('lnihmBundle');
  if (!root || root.getAttribute('data-lnihm-init')) return;
  root.setAttribute('data-lnihm-init', '1');
  var row = root.querySelector('.lnihm-bundle-row');
  var sub = root.querySelector('.lnihm-bundle-sub');
  var dotsWrap = root.querySelector('.lnihm-bundle-dots');

  var SRC = '/categories/bundle';
  var KEY = 'lnihm_bundle_cache_v4';

  var TTL = 720 * 60 * 1000;

  var MAX = 8;

  function money(n){ return 'NT$' + String(n).replace(/\B(?=(\d{3})+(?!\d))/g, ','); }
  function cleanTitle(t){ return t.replace(/\u3010[^\u3011]*\u3011\s*$/, '').trim(); }

  function makeCard(p){
    var a = document.createElement('a');
    a.className = 'lnihm-bundle-card';
    a.setAttribute('href', p.url);
    var photo = document.createElement('div');
    photo.className = 'lnihm-bundle-photo';
    if (p.img) photo.style.backgroundImage = "url('" + p.img.replace(/'/g, '%27') + "')";
    if (p.save > 0){
      var save = document.createElement('span');
      save.className = 'lnihm-bundle-save';
      save.textContent = '省 ' + money(p.save);
      photo.appendChild(save);
    }
    var body = document.createElement('div');
    body.className = 'lnihm-bundle-body';
    var title = document.createElement('div');
    title.className = 'lnihm-bundle-title';
    title.textContent = cleanTitle(p.title || '組合包');
    var prices = document.createElement('div');
    prices.className = 'lnihm-bundle-prices';
    var now = document.createElement('span');
    now.className = 'lnihm-bundle-now';
    now.textContent = money(p.sale);
    prices.appendChild(now);
    if (p.reg && p.reg > p.sale){
      var was = document.createElement('span');
      was.className = 'lnihm-bundle-was';
      was.textContent = money(p.reg);
      prices.appendChild(was);
    }

    body.appendChild(title); body.appendChild(prices);
    a.appendChild(photo); a.appendChild(body);
    return a;
  }

  var page = 0, pv = 1, pages = 1, lock = false;
  function cards(){ return [].slice.call(row.querySelectorAll('.lnihm-bundle-card')); }
  function step(cs){ return cs.length > 1 ? (cs[1].offsetLeft - cs[0].offsetLeft) : (cs[0] ? cs[0].offsetWidth : 1); }

  var twIv;
  function tween(el, to, dur, done){
    clearInterval(twIv); el.style.scrollSnapType = 'none';
    var start = el.scrollLeft, change = to - start, t0 = Date.now();
    if (Math.abs(change) < 1){ el.scrollLeft = to; el.style.scrollSnapType = ''; if (done) done(); return; }
    twIv = setInterval(function(){
      var p = Math.min(1, (Date.now() - t0) / dur);
      el.scrollLeft = start + change * (1 - Math.pow(1 - p, 3));
      if (p >= 1){ clearInterval(twIv); el.style.scrollSnapType = ''; if (done) done(); }
    }, 16);
  }
  function build(){
    var cs = cards(); if (!cs.length) return;
    var s = step(cs); pv = Math.max(1, Math.round(row.clientWidth / s));
    pages = Math.ceil(cs.length / pv);
    dotsWrap.innerHTML = '';
    if (pages > 1){
      for (var i = 0; i < pages; i++){
        var b = document.createElement('button');
        b.type = 'button'; b.setAttribute('aria-label', '第 ' + (i + 1) + ' 頁');
        (function(idx){ b.addEventListener('click', function(){ go(idx); }); })(i);
        dotsWrap.appendChild(b);
      }
    }
    page = Math.min(page, pages - 1); mark();
  }

  function pageLeft(p){
    var cs = cards(); if (!cs.length) return 0;
    var idx = Math.min(p * pv, cs.length - 1);
    var maxSc = row.scrollWidth - row.clientWidth;
    return Math.max(0, Math.min(cs[idx].offsetLeft - cs[0].offsetLeft, maxSc));
  }
  function nearestPage(){
    var best = 0, bd = Infinity;
    for (var i = 0; i < pages; i++){
      var d = Math.abs(row.scrollLeft - pageLeft(i));
      if (d < bd){ bd = d; best = i; }
    }
    return best;
  }

  var SPEED = 250;
  function go(p){
    p = Math.max(0, Math.min(p, pages - 1)); page = p;
    lock = true; tween(row, pageLeft(p), SPEED, function(){ lock = false; });
    mark();
  }

  var THIRD = 1 / 3;
  function snapByThird(){
    var cs = cards(); if (!cs.length) return;
    var s = step(cs), maxSc = row.scrollWidth - row.clientWidth, base = cs[0].offsetLeft;
    var delta = row.scrollLeft - dScroll;
    var a = Math.abs(delta), full = Math.floor(a / s), rem = a - full * s;
    var adv = full + (rem >= s * THIRD ? 1 : 0);
    var target = dScroll + (delta < 0 ? -1 : 1) * adv * s;
    target = Math.max(0, Math.min(target, maxSc));
    var best = target, bd = Infinity;
    for (var i = 0; i < cs.length; i++){
      var v = Math.max(0, Math.min(cs[i].offsetLeft - base, maxSc));
      var d = Math.abs(target - v);
      if (d < bd){ bd = d; best = v; }
    }
    lock = true;
    tween(row, best, SPEED, function(){ lock = false; page = nearestPage(); mark(); });
  }
  function mark(){ var k = dotsWrap.children; for (var i = 0; i < k.length; i++) k[i].setAttribute('aria-current', i === page ? 'true' : 'false'); }
  function onScroll(){
    if (lock || pages <= 1) return;
    var p = nearestPage();
    if (p !== page){ page = p; mark(); }
  }
  var st; row.addEventListener('scroll', function(){ clearTimeout(st); st = setTimeout(onScroll, 60); }, { passive: true });
  var rt; window.addEventListener('resize', function(){ clearTimeout(rt); rt = setTimeout(build, 150); });

  var dDown = false, dMoved = false, dX = 0, dScroll = 0;
  var DRAG_MIN = 8;
  function onMove(e){
    if (!dDown || e.pointerType !== 'mouse') return;
    var dx = e.clientX - dX;
    if (Math.abs(dx) > DRAG_MIN) dMoved = true;
    if (dMoved){ row.scrollLeft = dScroll - dx; e.preventDefault(); }
  }
  function dEnd(){
    if (!dDown) return; dDown = false;
    window.removeEventListener('pointermove', onMove);
    window.removeEventListener('pointerup', dEnd);
    window.removeEventListener('pointercancel', dEnd);
    row.classList.remove('lnihm-grabbing');

    if (dMoved) snapByThird();
    else row.style.scrollSnapType = '';
  }
  row.addEventListener('pointerdown', function(e){
    if (e.pointerType !== 'mouse' || e.button !== 0) return;
    dDown = true; dMoved = false; dX = e.clientX; dScroll = row.scrollLeft;
    row.style.scrollSnapType = 'none'; row.classList.add('lnihm-grabbing');
    window.addEventListener('pointermove', onMove);
    window.addEventListener('pointerup', dEnd);
    window.addEventListener('pointercancel', dEnd);
  });

  row.addEventListener('click', function(e){ if (dMoved){ e.preventDefault(); e.stopPropagation(); dMoved = false; } }, true);

  function render(list){
    if (!list || !list.length) return;
    row.innerHTML = '';
    var shown = list.slice(0, MAX);
    shown.forEach(function(p){ row.appendChild(makeCard(p)); });

    var maxSave = 0;
    shown.forEach(function(p){ if (p.save > maxSave) maxSave = p.save; });
    if (sub && maxSave > 0){

      sub.textContent = '成套帶回最高省 ' + money(maxSave);
    }
    page = 0; build();
  }

  function parse(html){
    var doc = new DOMParser().parseFromString(html, 'text/html');
    var items = doc.querySelectorAll('.productList__product');
    var seen = {}, out = [];
    for (var i = 0; i < items.length; i++){
      var el = items[i];
      var a = el.querySelector('a[href*="/products/"]');
      var t = el.querySelector('.title');
      var saleEl = el.querySelector('.price__sale');
      if (!a || !t || !saleEl) continue;
      var url = (a.getAttribute('href') || '').split('?')[0];
      if (!url || seen[url]) continue;
      var num = function(el2){
        if (!el2) return 0;
        var m = (el2.textContent || '').replace(/,/g, '').match(/(\d+)/);
        return m ? parseInt(m[1], 10) : 0;
      };
      var sale = num(saleEl);
      if (!sale) continue;
      var reg = num(el.querySelector('.price-crossed'));

      var img = '', imgEl = el.querySelector('img[data-srcset], img[srcset], img[data-src], img[src]');
      if (imgEl){
        var ss = imgEl.getAttribute('data-srcset') || imgEl.getAttribute('srcset') || '';
        var best = null, bestD = 1e9, re = /(https:\/\/[^\s,"]+?\/(\d+)x\.[a-z]+)\?/g, m2;
        while ((m2 = re.exec(ss)) !== null){
          var d = Math.abs(parseInt(m2[2], 10) - 720);
          if (d < bestD){ bestD = d; best = m2[1]; }
        }
        img = best || imgEl.getAttribute('data-src') || imgEl.getAttribute('src') || '';
      }
      if (!/^https:\/\//i.test(img)) continue;
      seen[url] = 1;
      out.push({ url: url, title: (t.textContent || '').trim(), sale: sale, reg: reg,
                 save: (reg && reg > sale) ? (reg - sale) : 0, img: img });
    }

    return out;
  }

  function refresh(){
    fetch(SRC, { credentials: 'same-origin' })
      .then(function(r){ return r.ok ? r.text() : Promise.reject(r.status); })
      .then(function(html){
        var list = parse(html);
        if (list.length){
          render(list);
          try { localStorage.setItem(KEY, JSON.stringify({ t: Date.now(), list: list.slice(0, MAX) })); } catch(e){}
        }
      })
      .catch(function(){   });
  }

  build();

  var usedCache = false;
  try {
    var c = JSON.parse(localStorage.getItem(KEY) || 'null');
    if (c && c.list && c.list.length){
      render(c.list);
      usedCache = true;
      if (Date.now() - c.t > TTL) refresh();
    }
  } catch(e){}
  if (!usedCache) refresh();
})();
</script>
