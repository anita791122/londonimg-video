    var cs = cells(); if (!cs.length) return 0;
    var idx = Math.min(p * pv, cs.length - 1);
    var maxSc = grid.scrollWidth - grid.clientWidth;
    return Math.max(0, Math.min(cs[idx].offsetLeft - cs[0].offsetLeft, maxSc));
  }
  function nearestPage(){
    var best = 0, bd = Infinity;
    for (var i = 0; i < pages; i++){ var d = Math.abs(grid.scrollLeft - pageLeft(i)); if (d < bd){ bd = d; best = i; } }
    return best;
  }
  function go(p){
    p = Math.max(0, Math.min(p, pages - 1)); page = p;
    lock = true; tween(grid, pageLeft(p), SPEED, function(){ lock = false; });
    mark();
  }
  function mark(){ var k = dotsWrap.children; for (var i = 0; i < k.length; i++) k[i].setAttribute('aria-current', i === page ? 'true' : 'false'); }
  function onScroll(){
    if (lock || pages <= 1) return;
    var p = nearestPage();
    if (p !== page){ page = p; mark(); }
  }

  var THIRD = 1 / 3;
  function snapByThird(){
    var cs = cells(); if (!cs.length) return;
    var s = step(cs), maxSc = grid.scrollWidth - grid.clientWidth, base = cs[0].offsetLeft;
    var delta = grid.scrollLeft - dScroll;
    var a = Math.abs(delta), full = Math.floor(a / s), rem = a - full * s;
    var adv = full + (rem >= s * THIRD ? 1 : 0);
    var target = dScroll + (delta < 0 ? -1 : 1) * adv * s;
    target = Math.max(0, Math.min(target, maxSc));
    var best = target, bd = Infinity;
    for (var i = 0; i < cs.length; i++){
      var v = Math.max(0, Math.min(cs[i].offsetLeft - base, maxSc));
      var d = Math.abs(target - v);
      if (d < bd){ bd = d; best = v; }
    }
    lock = true; tween(grid, best, SPEED, function(){ lock = false; page = nearestPage(); mark(); });
  }
  var stmr; grid.addEventListener('scroll', function(){ clearTimeout(stmr); stmr = setTimeout(onScroll, 60); }, { passive: true });
  var rtmr; window.addEventListener('resize', function(){ clearTimeout(rtmr); rtmr = setTimeout(layout, 150); });

  var dDown=false,dMoved=false,dX=0,dScroll=0;
  var DRAG_MIN=8;
  function onMove(e){
    if(!dDown||e.pointerType!=='mouse')return;
    var dx=e.clientX-dX;
    if(Math.abs(dx)>DRAG_MIN)dMoved=true;
    if(dMoved){ grid.scrollLeft=dScroll-dx; e.preventDefault(); }
  }
  function dEnd(){
    if(!dDown)return;dDown=false;
    window.removeEventListener('pointermove',onMove);
    window.removeEventListener('pointerup',dEnd);
    window.removeEventListener('pointercancel',dEnd);
    grid.classList.remove('lnihm-grabbing');

    if(dMoved) snapByThird();
    else grid.style.scrollSnapType='';
  }
  grid.addEventListener('pointerdown',function(e){
    if(e.pointerType!=='mouse'||e.button!==0)return;
    dDown=true;dMoved=false;dX=e.clientX;dScroll=grid.scrollLeft;
    grid.style.scrollSnapType='none';grid.classList.add('lnihm-grabbing');
    window.addEventListener('pointermove',onMove);
    window.addEventListener('pointerup',dEnd);
    window.addEventListener('pointercancel',dEnd);
  });

  grid.addEventListener('click',function(e){ if(dMoved){e.preventDefault();e.stopPropagation();dMoved=false;} },true);

  function render(tiles){
    if (!tiles || !tiles.length) return;
    grid.innerHTML = '';
    tiles.slice(0, MAX).forEach(function(t){ grid.appendChild(makeCell(t)); });
    page = 0; layout();
  }

  function parse(html){
    var doc = new DOMParser().parseFromString(html, 'text/html');
    var anchors = doc.querySelectorAll('a[href*="instagram.com"]');
    var seen = {}, tiles = [];
    for (var i = 0; i < anchors.length; i++){
      var a = anchors[i];
      var img = a.querySelector('img');
      if (!img) continue;
      var ig = (a.getAttribute('href') || '').trim();
      var src = (img.getAttribute('data-src') || img.getAttribute('src') || '').trim();
      if (!safeIg(ig) || !safeImg(src)) continue;
      if (seen[ig]) continue;
      if (!KOL[shortcode(ig)]) continue;
      seen[ig] = 1;
      tiles.push({ ig: ig, img: src });
    }
    return tiles;
  }

  function refresh(){
    fetch(SRC, { credentials: 'same-origin' })
      .then(function(r){ return r.ok ? r.text() : Promise.reject(r.status); })
      .then(function(html){
        var tiles = parse(html);
        if (tiles.length){
          render(tiles);
          try { localStorage.setItem(KEY, JSON.stringify({ t: Date.now(), tiles: tiles })); } catch(e){}
        }
      })
      .catch(function(){   });
  }

  function decorateFallback(){
    cells().forEach(function(cell){
      if (cell.querySelector('.lnihm-kol-tags')) return;
      var a = cell.querySelector('.lnihm-kol-link');
      if (a) addTags(cell, KOL[shortcode(a.getAttribute('href'))]);
    });
  }

  decorateFallback();
  layout();

  var usedCache = false;
  try {
    var c = JSON.parse(localStorage.getItem(KEY) || 'null');
    if (c && c.tiles && c.tiles.length){
      render(c.tiles);
      usedCache = true;
      if (Date.now() - c.t > TTL) refresh();
    }
  } catch(e){}
  if (!usedCache) refresh();
})();
</script>
